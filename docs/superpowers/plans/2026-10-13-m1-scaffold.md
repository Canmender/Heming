# Heming M1（骨架）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 搭起 pnpm monorepo 四包骨架，完成账号/好友/1:1 会话的云端 API + WebSocket、三栏导航壳（白底浅灰分割线视觉规范）与主页/工作区/文件树基础界面，产出可运行、可测的端到端最小系统。

**Architecture:** 云端中心化——`server`（Express 5 + ws + 内存存储）是唯一数据源；`web`（Vite + React 18）为薄客户端，经 token 鉴权的 REST + WS 交互；`shared` 提供两端共用的 Zod 协议；`desktop` 是 Electron 壳。M1 存储层为内存实现（接口已定义，M2 落 SQLite），使测试无需文件数据库。

**Tech Stack:** Node ≥20、pnpm ≥10、TypeScript 5、Express 5、ws、Zod、React 18、Vite 5、Tailwind CSS 3、Zustand、Vitest 1、Supertest、Electron（壳）。

## Global Constraints

- 禁止在任何 UI 中使用 emoji 图标；图标一律 1.5px stroke 内联 SVG。
- 分割线/边框色用 `#e6e6e6`（面板内侧 `#ebebeb`），禁止纯黑分割线；纯黑 `#1a1a1a` 仅作文字与小面积强调。
- 两端通信协议只定义在 `shared`，服务端与前端均通过 Zod schema 校验，禁止任一侧私造消息格式。
- 密码只存 scrypt 哈希；token 经 `Authorization: Bearer` 或 WS 查询参数 `?token=` 传递，明文 Key 不进库。
- 所有新代码必须有对应 Vitest 测试；测试命令统一 `pnpm --filter <pkg> test`。
- 提交信息遵循 `feat:/test:/chore:/docs:` 前缀，每个任务至少一次提交。
- M1 不接任何真实 LLM；不做 E2EE、语音、表情、转发（属二期）。

```ts
import { z } from "zod";

export const UserSchema = z.object({
  id: z.string().min(1),
  username: z.string().min(3).max(32),
  displayName: z.string().min(1).max(64),
  avatar: z.string().max(512).optional(),
});
export type User = z.infer<typeof UserSchema>;

export const MessagePayloadSchema = z.object({ text: z.string().max(20000) });
export type MessagePayload = z.infer<typeof MessagePayloadSchema>;

export const MessageSchema = z.object({
  id: z.string().min(1),
  conversationId: z.string().min(1),
  senderType: z.enum(["user", "agent"]),
  senderId: z.string().min(1),
  payload: MessagePayloadSchema,
  replyTo: z.string().nullable(),
  createdAt: z.string(),
});
export type Message = z.infer<typeof MessageSchema>;

export const WsClientEventSchema = z.discriminatedUnion("type", [
  z.object({
    type: z.literal("send_message"),
    conversationId: z.string().min(1),
    payload: MessagePayloadSchema,
    replyTo: z.string().optional(),
  }),
  z.object({
    type: z.literal("mark_read"),
    conversationId: z.string().min(1),
  }),
]);
export type WsClientEvent = z.infer<typeof WsClientEventSchema>;

export const WsServerEventSchema = z.discriminatedUnion("type", [
  z.object({ type: z.literal("message_new"), message: MessageSchema }),
  z.object({
    type: z.literal("read_update"),
    conversationId: z.string().min(1),
    userId: z.string().min(1),
    at: z.string(),
  }),
  z.object({
    type: z.literal("error"),
    code: z.string().min(1),
    message: z.string().min(1),
  }),
]);
export type WsServerEvent = z.infer<typeof WsServerEventSchema>;

export const RegisterBodySchema = z.object({
  username: z.string().min(3).max(32).regex(/^[a-zA-Z0-9_]+$/),
  password: z.string().min(6).max(128),
  displayName: z.string().min(1).max(64),
});
export type RegisterBody = z.infer<typeof RegisterBodySchema>;

export const LoginBodySchema = z.object({
  username: z.string().min(3).max(32),
  password: z.string().min(6).max(128),
});
export type LoginBody = z.infer<typeof LoginBodySchema>;

export const CreateConversationBodySchema = z.object({ peerId: z.string().min(1) });
export const CreateFriendRequestBodySchema = z.object({ username: z.string().min(3).max(32) });

export const SharedFileMetaSchema = z.object({
  id: z.string().min(1),
  owner: z.string().min(1),
  name: z.string().min(1),
  mime: z.string().min(1),
  size: z.number().int().nonnegative(),
  updatedAt: z.string(),
});
export type SharedFileMeta = z.infer<typeof SharedFileMetaSchema>;
```

`packages/server/src/store.ts`:

```ts
import { randomUUID } from "node:crypto";
import type { User, Message } from "@heming/shared";

export interface FriendshipRequest {
  id: string;
  fromId: string;
  toId: string;
  status: "pending" | "accepted";
}

export interface Store {
  createUser(username: string, passwordHash: string, displayName: string): User;
  findUserByUsername(username: string): { user: User; passwordHash: string } | null;
  findUserById(id: string): User | null;
  createFriendRequest(fromId: string, toUsername: string): { id: string; from: User; to: User; status: "pending" };
  acceptFriendRequest(requestId: string, userId: string): { id: string; user: User };
  listFriends(userId: string): { id: string; user: User }[];
  listIncomingRequests(userId: string): { id: string; from: User }[];
  createDirectConversation(aId: string, bId: string): { id: string; type: "direct"; peer: User };
  listConversations(userId: string): { id: string; type: "direct"; peer: User }[];
  isMember(conversationId: string, userId: string): boolean;
  addMessage(
    conversationId: string,
    senderType: "user" | "agent",
    senderId: string,
    payload: { text: string },
    replyTo: string | null,
  ): Message;
  listMessages(conversationId: string, limit?: number): Message[];
}

export function createMemoryStore(): Store {
  const users = new Map<string, { user: User; passwordHash: string }>();
  const usersByName = new Map<string, string>();
  const requests: FriendshipRequest[] = [];
  const conversations = new Map<string, { members: string[] }>();
  const messages = new Map<string, Message[]>();
  const pairToConv = new Map<string, string>();

  const mustUser = (id: string): User => {
    const row = users.get(id);
    if (!row) throw new Error("user not found");
    return row.user;
  };
  const pairKey = (a: string, b: string): string => [a, b].sort().join("|");

  return {
    createUser(username, passwordHash, displayName) {
      if (usersByName.has(username)) throw new Error("username taken");
      const user: User = { id: randomUUID(), username, displayName };
      users.set(user.id, { user, passwordHash });
      usersByName.set(username, user.id);
      return user;
    },
    findUserByUsername(username) {
      const id = usersByName.get(username);
      return id ? users.get(id)! : null;
    },
    findUserById(id) {
      return users.get(id)?.user ?? null;
    },
    createFriendRequest(fromId, toUsername) {
      mustUser(fromId);
      const toId = usersByName.get(toUsername);
      if (!toId) throw new Error("peer not found");
      if (toId === fromId) throw new Error("cannot friend self");
      const existing = requests.find(
        (r) =>
          r.status === "pending" &&
          ((r.fromId === fromId && r.toId === toId) || (r.fromId === toId && r.toId === fromId)),
      );
      if (existing) throw new Error("request already exists");
      const req: FriendshipRequest = { id: randomUUID(), fromId, toId, status: "pending" };
      requests.push(req);
      return { id: req.id, from: mustUser(fromId), to: mustUser(toId), status: "pending" };
    },
    acceptFriendRequest(requestId, userId) {
      const req = requests.find((r) => r.id === requestId);
      if (!req) throw new Error("request not found");
      if (req.toId !== userId) throw new Error("not request recipient");
      req.status = "accepted";
      return { id: req.id, user: mustUser(req.fromId) };
    },
    listFriends(userId) {
      return requests
        .filter((r) => r.status === "accepted" && (r.fromId === userId || r.toId === userId))
        .map((r) => ({ id: r.id, user: mustUser(r.fromId === userId ? r.toId : r.fromId) }));
    },
    listIncomingRequests(userId) {
      return requests
        .filter((r) => r.status === "pending" && r.toId === userId)
        .map((r) => ({ id: r.id, from: mustUser(r.fromId) }));
    },
    createDirectConversation(aId, bId) {
      mustUser(aId);
      mustUser(bId);
      const friends = requests.some(
        (r) => r.status === "accepted" && pairKey(r.fromId, r.toId) === pairKey(aId, bId),
      );
      if (!friends) throw new Error("not friends");
      const key = pairKey(aId, bId);
      const existingId = pairToConv.get(key);
      if (existingId) {
        return { id: existingId, type: "direct", peer: mustUser(bId) };
      }
      const id = randomUUID();
      conversations.set(id, { members: [aId, bId] });
      messages.set(id, []);
      pairToConv.set(key, id);
      return { id, type: "direct", peer: mustUser(bId) };
    },
    listConversations(userId) {
      const out: { id: string; type: "direct"; peer: User }[] = [];
      for (const [id, conv] of conversations) {
        if (!conv.members.includes(userId)) continue;
        const peerId = conv.members.find((m) => m !== userId);
        if (peerId) out.push({ id, type: "direct", peer: mustUser(peerId) });
      }
      return out;
    },
    isMember(conversationId, userId) {
      return conversations.get(conversationId)?.members.includes(userId) ?? false;
    },
    addMessage(conversationId, senderType, senderId, payload, replyTo) {
      if (!conversations.has(conversationId)) throw new Error("conversation not found");
      const msg: Message = {
        id: randomUUID(),
        conversationId,
        senderType,
        senderId,
        payload,
        replyTo,
        createdAt: new Date().toISOString(),
      };
      messages.get(conversationId)!.push(msg);
      return msg;
    },
    listMessages(conversationId, limit) {
      const all = messages.get(conversationId);
      if (!all) throw new Error("conversation not found");
      return limit ? all.slice(-limit) : [...all];
    },
  };
}
```

```ts
import { randomBytes, createHash } from "node:crypto";

const tokens = new Map<string, string>(); // token -> userId

export function issueToken(userId: string): string {
  const token = randomBytes(32).toString("hex");
  tokens.set(token, userId);
  return token;
}

export function userIdFromToken(token: string): string | null {
  if (!token) return null;
  const digest = createHash("sha256").update(token).digest("hex");
  for (const [t, id] of tokens) {
    if (createHash("sha256").update(t).digest("hex") === digest) return id;
  }
  return null;
}
```

```ts
import express, { type Request, type Response, type NextFunction } from "express";
import { scryptSync, randomBytes, timingSafeEqual } from "node:crypto";
import {
  RegisterBodySchema,
  LoginBodySchema,
  CreateConversationBodySchema,
  CreateFriendRequestBodySchema,
} from "@heming/shared";
import type { Store } from "./store.js";
import { issueToken, userIdFromToken } from "./auth.js";

export function hashPassword(password: string): string {
  const salt = randomBytes(16).toString("hex");
  const hash = scryptSync(password, salt, 64).toString("hex");
  return `${salt}:${hash}`;
}

export function verifyPassword(password: string, stored: string): boolean {
  const [salt, hash] = stored.split(":");
  if (!salt || !hash) return false;
  const candidate = scryptSync(password, salt, 64);
  const expected = Buffer.from(hash, "hex");
  return candidate.length === expected.length && timingSafeEqual(candidate, expected);
}

function authOnly(req: Request, res: Response, next: NextFunction): void {
  const header = req.header("authorization") ?? "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : "";
  const userId = userIdFromToken(token);
  if (!userId) {
    res.status(401).json({ error: "unauthorized" });
    return;
  }
  (req as Request & { userId: string }).userId = userId;
  next();
}

function userId(req: Request): string {
  return (req as Request & { userId: string }).userId;
}

export function createApp(store: Store): express.Express {
  const app = express();
  app.use(express.json({ limit: "1mb" }));

  app.get("/api/health", (_req, res) => {
    res.json({ ok: true });
  });

  app.post("/api/auth/register", (req, res) => {
    const parsed = RegisterBodySchema.safeParse(req.body);
    if (!parsed.success) {
      res.status(400).json({ error: parsed.error.issues[0]?.message ?? "invalid body" });
      return;
    }
    const { username, password, displayName } = parsed.data;
    try {
      const user = store.createUser(username, hashPassword(password), displayName);
      res.status(201).json({ token: issueToken(user.id), user });
    } catch (e) {
      res.status(400).json({ error: (e as Error).message });
    }
  });

  app.post("/api/auth/login", (req, res) => {
    const parsed = LoginBodySchema.safeParse(req.body);
    if (!parsed.success) {
      res.status(400).json({ error: "invalid body" });
      return;
    }
    const row = store.findUserByUsername(parsed.data.username);
    if (!row || !verifyPassword(parsed.data.password, row.passwordHash)) {
      res.status(401).json({ error: "invalid credentials" });
      return;
    }
    res.json({ token: issueToken(row.user.id), user: row.user });
  });

  app.get("/api/me", authOnly, (req, res) => {
    const user = store.findUserById(userId(req));
    if (!user) {
      res.status(401).json({ error: "unauthorized" });
      return;
    }
    res.json(user);
  });

  app.get("/api/friends", authOnly, (req, res) => {
    res.json(store.listFriends(userId(req)));
  });

  app.get("/api/friends/requests", authOnly, (req, res) => {
    res.json(store.listIncomingRequests(userId(req)));
  });

  app.post("/api/friends/request", authOnly, (req, res) => {
    const parsed = CreateFriendRequestBodySchema.safeParse(req.body);
    if (!parsed.success) {
      res.status(400).json({ error: "invalid body" });
      return;
    }
    try {
      const result = store.createFriendRequest(userId(req), parsed.data.username);
      res.status(201).json(result);
    } catch (e) {
      res.status(400).json({ error: (e as Error).message });
    }
  });

  app.post("/api/friends/:id/accept", authOnly, (req, res) => {
    try {
      res.json(store.acceptFriendRequest(req.params.id, userId(req)));
    } catch (e) {
      res.status(400).json({ error: (e as Error).message });
    }
  });

  app.get("/api/conversations", authOnly, (req, res) => {
    res.json(store.listConversations(userId(req)));
  });

  app.post("/api/conversations", authOnly, (req, res) => {
    const parsed = CreateConversationBodySchema.safeParse(req.body);
    if (!parsed.success) {
      res.status(400).json({ error: "invalid body" });
      return;
    }
    try {
      res.status(201).json(store.createDirectConversation(userId(req), parsed.data.peerId));
    } catch (e) {
      res.status(400).json({ error: (e as Error).message });
    }
  });

  app.get("/api/conversations/:id/messages", authOnly, (req, res) => {
    if (!store.isMember(req.params.id, userId(req))) {
      res.status(403).json({ error: "forbidden" });
      return;
    }
    res.json(store.listMessages(req.params.id));
  });

  app.get("/api/files", authOnly, (_req, res) => {
    res.json([]);
  });

  return app;
}
```

```ts
import type { Server as HttpServer } from "node:http";
import { WebSocketServer, WebSocket } from "ws";
import { WsClientEventSchema, type WsServerEvent } from "@heming/shared";
import type { Store } from "./store.js";
import { userIdFromToken } from "./auth.js";

export function attachWs(server: HttpServer, store: Store): WebSocketServer {
  const wss = new WebSocketServer({ noServer: true });
  const socketsByUser = new Map<string, Set<WebSocket>>();

  server.on("upgrade", (req, socket, head) => {
    const url = new URL(req.url ?? "/", "http://localhost");
    if (url.pathname !== "/api/ws") {
      socket.destroy();
      return;
    }
    const userId = userIdFromToken(url.searchParams.get("token") ?? "");
    if (!userId) {
      socket.destroy();
      return;
    }
    wss.handleUpgrade(req, socket, head, (ws) => {
      wss.emit("connection", ws, req, userId);
    });
  });

  function send(ws: WebSocket, event: WsServerEvent): void {
    if (ws.readyState === WebSocket.OPEN) ws.send(JSON.stringify(event));
  }

  function broadcastToConversation(conversationId: string, event: WsServerEvent): void {
    for (const [userId, sockets] of socketsByUser) {
      if (!store.isMember(conversationId, userId)) continue;
      for (const ws of sockets) send(ws, event);
    }
  }

  wss.on("connection", (ws: WebSocket, _req, userId: string) => {
    let sockets = socketsByUser.get(userId);
    if (!sockets) {
      sockets = new Set();
      socketsByUser.set(userId, sockets);
    }
    sockets.add(ws);

    ws.on("message", (raw) => {
      let json: unknown;
      try {
        json = JSON.parse(raw.toString());
      } catch {
        send(ws, { type: "error", code: "bad_event", message: "invalid json" });
        return;
      }
      const parsed = WsClientEventSchema.safeParse(json);
      if (!parsed.success) {
        send(ws, { type: "error", code: "bad_event", message: "invalid event" });
        return;
      }
      const event = parsed.data;
      if (!store.isMember(event.conversationId, userId)) {
        send(ws, { type: "error", code: "forbidden", message: "not a member" });
        return;
      }
      if (event.type === "send_message") {
        const message = store.addMessage(
          event.conversationId,
          "user",
          userId,
          { text: event.payload.text },
          event.replyTo ?? null,
        );
        broadcastToConversation(event.conversationId, { type: "message_new", message });
      } else {
        broadcastToConversation(event.conversationId, {
          type: "read_update",
          conversationId: event.conversationId,
          userId,
          at: new Date().toISOString(),
        });
      }
    });

    ws.on("close", () => {
      sockets!.delete(ws);
      if (sockets!.size === 0) socketsByUser.delete(userId);
    });

    ws.on("error", () => { /* 连接级错误忽略，避免未捕获异常崩溃进程 */ });
  });

  return wss;
}
```

```ts
import { createServer, type Server as HttpServer } from "node:http";
import { createApp } from "./app.js";
import { createMemoryStore, type Store } from "./store.js";
import { attachWs } from "./ws.js";

export interface ServerContext {
  server: HttpServer;
  store: Store;
  port: number;
  close(): Promise<void>;
}

export async function startServer(port: number): Promise<ServerContext> {
  const store = createMemoryStore();
  const app = createApp(store);
  const server = createServer(app);
  attachWs(server, store);
  await new Promise<void>((resolve) => server.listen(port, "127.0.0.1", resolve));
  const address = server.address();
  const actualPort = typeof address === "object" && address ? address.port : port;
  return {
    server,
    store,
    port: actualPort,
    close: () =>
      new Promise((resolve, reject) => {
        server.close((err) => (err ? reject(err) : resolve()));
      }),
  };
}

const isDirectRun = process.argv[1]?.endsWith("index.ts") || process.argv[1]?.endsWith("index.js");
if (isDirectRun) {
  const port = Number(process.env.PORT ?? 8787);
  startServer(port).then((ctx) => {
    console.log(`heming server listening on http://127.0.0.1:${ctx.port}`);
  });
}
```

```ts
interface UiState {
  page: Page;
  navCollapsed: boolean;
  activeConversationId: string | null;
  setPage: (page: Page) => void;
  setNavCollapsed: (v: boolean) => void;

export const useUiStore = create<UiState>((set, get) => ({

  navCollapsed: false,
  activeConversationId: null,
  setPage: (page) => set({ page, navCollapsed: false, activeConversationId: null }),
  setNavCollapsed: (v) => set({ navCollapsed: v }),
```

```tsx
import { useState } from "react";
import type { SharedFileMeta } from "@heming/shared";
import { ChevronIcon, FileIcon, FolderIcon } from "./icons.js";

export type TreeItem = SharedFileMeta & { parentId?: string };

interface Props {
  items: TreeItem[];
  localPaths: string[];
}

function Row({ label, depth, onClick, ariaLabel, expandable, expanded }: {
  label: string;
  depth: number;
  onClick?: () => void;
  ariaLabel?: string;
  expandable?: boolean;
  expanded?: boolean;
}) {
  return (
    <button
      type="button"
      aria-label={ariaLabel}
      onClick={onClick}
      className="flex w-full items-center gap-1.5 rounded px-2 py-[3px] text-left text-[12px] text-[#444] hover:bg-[#f3f3f3]"
      style={{ paddingLeft: 8 + depth * 14 }}
    >
      {expandable ? (
        <ChevronIcon
          width={11}
          height={11}
          className={`shrink-0 text-muted transition-transform ${expanded ? "rotate-90" : ""}`}
        />
      ) : (
        <span className="w-[11px] shrink-0" />
      )}
      <span className="truncate">{label}</span>
    </button>
  );
}

function SectionRows({ title, items, rootParentId }: { title: string; items: TreeItem[]; rootParentId?: string }) {
  const [open, setOpen] = useState<Set<string>>(new Set());
  const toggle = (id: string) =>
    setOpen((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });

  const children = (id: string) => items.filter((i) => i.parentId === id);
  const roots = rootParentId === undefined ? items.filter((i) => !i.parentId) : children(rootParentId);

  const render = (list: TreeItem[], depth: number): JSX.Element[] =>
    list.map((item) => {
      const isDir = item.mime === "inode/directory";
      const kids = isDir ? children(item.id) : [];
      const isOpen = open.has(item.id);
      return (
        <div key={item.id}>
          <Row
            label={item.name}
            depth={depth}
            ariaLabel={isDir ? `切换文件夹 ${item.name}` : undefined}
            expandable={isDir}
            expanded={isOpen}
            onClick={isDir ? () => toggle(item.id) : undefined}
          />
          {isDir && isOpen ? render(kids, depth + 1) : null}
        </div>
      );
    });

  return (
    <div className="mb-3">
      <div className="px-2 pb-1 text-[10px] font-bold uppercase tracking-wider text-muted">{title}</div>
      {roots.length > 0 ? render(roots, 0) : <div className="px-2 text-[11px] text-muted">（空）</div>}
    </div>
  );
}

export function FileTree({ items, localPaths }: Props) {
  return (
    <div className="flex h-full flex-col gap-1 overflow-y-auto bg-panel py-2">
      <SectionRows title="云端共享" items={items} />
      <div className="mx-2 border-t border-line-soft" />
      <div className="mb-1 mt-1 px-2 pb-0 text-[10px] font-bold uppercase tracking-wider text-muted">
        本地工作目录
      </div>
      {localPaths.length === 0 ? (
        <div className="px-2 text-[11px] text-muted">未绑定本地目录</div>
      ) : (
        localPaths.map((p) => (
          <Row key={p} label={p.replace(/\\/g, "/").split("/").filter(Boolean).pop() ?? p} depth={0} />
        ))
      )}
    </div>
  );
}
```

```tsx
import { useEffect } from "react";
import { useUiStore } from "./store/ui.js";
import { NavSidebar } from "./components/NavSidebar.js";
import { NavRail } from "./components/NavRail.js";
import { FileTree, type TreeItem } from "./components/FileTree.js";

export function App() {
  const page = useUiStore((s) => s.page);
  const navCollapsed = useUiStore((s) => s.navCollapsed);
  const setNavCollapsed = useUiStore((s) => s.setNavCollapsed);
  const setPage = useUiStore((s) => s.setPage);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if ((e.metaKey || e.ctrlKey) && e.key === "\\") {
        e.preventDefault();
        setNavCollapsed(!navCollapsed);
      }
    };
    window.addEventListener("keydown", onKey);

          : page === "memory"
            ? "记忆区（全局记忆 + 知识库）二期开放"
            : "内容随后续任务填充"}

      {page !== "chat" ? null : (
        <button
          type="button"
          className="mt-4 w-fit rounded-lg border border-line px-3 py-1.5 text-[13px] shadow-card"

      <aside className="w-[240px] shrink-0 border-l border-line-soft">
        <FileTree items={demoItems} localPaths={[]} />
```

```ts
let token: string | null = localStorage.getItem("heming.token");

export function setToken(t: string | null): void {
  token = t;
  if (t) localStorage.setItem("heming.token", t);
  else localStorage.removeItem("heming.token");
}

export function getToken(): string | null {
  return token;
}

export class ApiError extends Error {
  status: number;
  constructor(status: number, message: string) {
    super(message);
    this.status = status;
  }
}

export async function api<T>(path: string, init: RequestInit = {}): Promise<T> {
  const headers = new Headers(init.headers);
  headers.set("Content-Type", "application/json");
  if (token) headers.set("Authorization", `Bearer ${token}`);
  const res = await fetch(path, { ...init, headers });
  const body = (await res.json().catch(() => ({}))) as { error?: string };
  if (!res.ok) throw new ApiError(res.status, body.error ?? `http ${res.status}`);
  return body as T;
}
```

```ts
import { create } from "zustand";
import { login as apiLogin, register as apiRegister, fetchMe, setToken, getToken } from "../lib/api.js";
import type { User } from "@heming/shared";

interface AuthState {
  token: string | null;
  user: User | null;
  status: "idle" | "pending" | "authed" | "error";
  error: string | null;
  login: (username: string, password: string) => Promise<void>;
  register: (username: string, password: string, displayName: string) => Promise<void>;
  logout: () => void;
  restore: () => Promise<void>;
}

export const useAuthStore = create<AuthState>((set) => ({
  token: getToken(),
  user: null,
  status: "idle",
  error: null,
  async login(username, password) {
    set({ status: "pending", error: null });
    try {
      const { token, user } = await apiLogin(username, password);
      setToken(token);
      set({ token, user, status: "authed" });
    } catch (e) {
      set({ status: "error", error: (e as Error).message });
    }
  },
  async register(username, password, displayName) {
    set({ status: "pending", error: null });
    try {
      const { token, user } = await apiRegister(username, password, displayName);
      setToken(token);
      set({ token, user, status: "authed" });
    } catch (e) {
      set({ status: "error", error: (e as Error).message });
    }
  },
  logout() {
    setToken(null);
    set({ token: null, user: null, status: "idle", error: null });
  },
  async restore() {
    if (!getToken()) return;
    set({ status: "pending" });
    try {
      const user = await fetchMe();
      set({ user, token: getToken(), status: "authed" });
    } catch {
      setToken(null);
      set({ token: null, user: null, status: "idle" });
    }
  },
}));
```

```ts
import { create } from "zustand";
import type { Message, User } from "@heming/shared";
import {
  fetchConversations,
  fetchFriends,
  fetchIncomingRequests,
  createConversation,
  fetchMessages,
} from "../lib/api.js";

interface ChatState {
  conversations: { id: string; type: "direct"; peer: User }[];
  friends: { id: string; user: User }[];
  incoming: { id: string; from: User }[];
  messages: Record<string, Message[]>;
  loadWorkspace: () => Promise<void>;
  openDirect: (peerId: string) => Promise<string>;
  appendMessage: (m: Message) => void;
  setMessages: (convId: string, msgs: Message[]) => void;
}

export const useChatStore = create<ChatState>((set, get) => ({
  conversations: [],
  friends: [],
  incoming: [],
  messages: {},
  async loadWorkspace() {
    const [conversations, friends, incoming] = await Promise.all([
      fetchConversations(),
      fetchFriends(),
      fetchIncomingRequests(),
    ]);
    set({ conversations, friends, incoming });
  },
  async openDirect(peerId) {
    const conv = await createConversation(peerId);
    if (!get().conversations.some((c) => c.id === conv.id)) {
      set({ conversations: [...get().conversations, conv] });
    }
    await get().loadWorkspace();
    return conv.id;
  },
  appendMessage(m) {
    const list = get().messages[m.conversationId] ?? [];
    if (list.some((x) => x.id === m.id)) return;
    set({ messages: { ...get().messages, [m.conversationId]: [...list, m] } });
  },
  setMessages(convId, msgs) {
    // 历史响应落定前可能已有 message_new 入列（fetch 与 WS 交错）：
    // 以历史为底，保留本地独有的条目（按 id 去重，按 createdAt 插回原时间位）
    const localOnly = (get().messages[convId] ?? []).filter(
      (m) => !msgs.some((x) => x.id === m.id),
    );
    const merged = [...msgs];
    for (const m of localOnly) {
      const at = merged.findIndex((x) => x.createdAt > m.createdAt);
      if (at === -1) merged.push(m);
      else merged.splice(at, 0, m);
    }
    set({ messages: { ...get().messages, [convId]: merged } });
  },
}));
```

`packages/web/src/lib/ws.ts`:

```ts
import type { WsClientEvent, WsServerEvent } from "@heming/shared";
import { getToken } from "./api.js";

export interface WsHandlers {
  onEvent: (e: WsServerEvent) => void;
  onOpen?: () => void;
  onClose?: () => void;
}

export interface WsHandle {
  send: (payload: WsClientEvent) => void;
  close: () => void;
}

export function connectWs(handlers: WsHandlers): WsHandle {
  let ws: WebSocket | null = null;
  let closed = false;
  let attempt = 0;
  let timer: ReturnType<typeof setTimeout> | null = null;
  const queue: string[] = [];

  const open = () => {
    if (closed) return;
    const token = getToken();
    if (!token) return;
    const proto = location.protocol === "https:" ? "wss" : "ws";
    ws = new WebSocket(`${proto}://${location.host}/api/ws?token=${token}`);
    ws.onopen = () => {
      attempt = 0;
      handlers.onOpen?.();
      for (const frame of queue.splice(0)) ws?.send(frame);
    };
    ws.onmessage = (e) => {
      try {
        handlers.onEvent(JSON.parse(e.data) as WsServerEvent);
      } catch {
        /* 忽略无法解析的帧 */
      }
    };
    ws.onclose = () => {
      handlers.onClose?.();
      if (closed) return;
      const delay = Math.min(1000 * 2 ** attempt, 30_000);
      attempt += 1;
      timer = setTimeout(open, delay);
    };
  };

  open();

  return {
    send(payload) {
      const frame = JSON.stringify(payload);
      if (ws?.readyState === WebSocket.OPEN) ws.send(frame);
      else queue.push(frame);
    },
    close() {
      closed = true;
      if (timer) clearTimeout(timer);
      ws?.close();
    },
  };
}
```

```ts
import { create } from "zustand";

const STORAGE_KEY = "heming.tokenUsage";
const MAX_RECORDS = 5000;

export interface UsageRecord {
  model: string;
  tokens: number;
  at: string;
}

export function monthKeyOf(iso: string): string {
  const d = new Date(iso);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
}

export function dayKeyOf(iso: string): string {
  const d = new Date(iso);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(
    d.getDate(),
  ).padStart(2, "0")}`;
}

export function summarizeMonth(
  records: UsageRecord[],
  monthKey: string,
): { total: number; byModel: Record<string, number> } {
  let total = 0;
  const byModel: Record<string, number> = {};
  for (const r of records) {
    if (monthKeyOf(r.at) !== monthKey) continue;
    total += r.tokens;
    byModel[r.model] = (byModel[r.model] ?? 0) + r.tokens;
  }
  return { total, byModel };
}

export function daysOfMonth(monthKey: string): string[] {
  const [year, month] = monthKey.split("-").map(Number);
  const count = new Date(year, month, 0).getDate();
  return Array.from({ length: count }, (_, i) => `${monthKey}-${String(i + 1).padStart(2, "0")}`);
}

export function dailyTotals(
  records: UsageRecord[],
  monthKey: string,
): { day: string; total: number; byModel: Record<string, number> }[] {
  const byDay = new Map<string, { total: number; byModel: Record<string, number> }>();
  for (const r of records) {
    if (monthKeyOf(r.at) !== monthKey) continue;
    const day = dayKeyOf(r.at);
    const entry = byDay.get(day) ?? { total: 0, byModel: {} };
    entry.total += r.tokens;
    entry.byModel[r.model] = (entry.byModel[r.model] ?? 0) + r.tokens;
    byDay.set(day, entry);
  }
  return daysOfMonth(monthKey).map((day) => {
    const hit = byDay.get(day);
    return { day, total: hit?.total ?? 0, byModel: hit?.byModel ?? {} };
  });
}

function load(): UsageRecord[] {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (raw) {
      const parsed = JSON.parse(raw) as { records?: UsageRecord[] };
      return Array.isArray(parsed.records) ? parsed.records : [];
    }
  } catch {
    /* 损坏数据回退空日志 */
  }
  return [];
}

function save(records: UsageRecord[]): void {
  localStorage.setItem(STORAGE_KEY, JSON.stringify({ records }));
}

interface TokenState {
  quota: number;
  records: UsageRecord[];
  record: (model: string, tokens: number, at?: string) => void;
  reset: () => void;
}

export const useTokenUsageStore = create<TokenState>((set, get) => ({
  quota: 3_000_000,
  records: load(),
  record(model, tokens, at) {
    const next = [...get().records, { model, tokens, at: at ?? new Date().toISOString() }].slice(
      -MAX_RECORDS,
    );
    save(next);
    set({ records: next });
  },
  reset() {
    save([]);
    set({ records: [] });
  },
}));
```

```ts
interface StoredFile {
  meta: SharedFileMeta;
  content: string; // base64
}
const files = new Map<string, StoredFile>();

app.get("/api/files", authOnly, (_req, res) => {
  res.json([...files.values()].map((f) => f.meta));
});

app.post("/api/files", authOnly, (req, res) => {
  const parsed = CreateFileBodySchema.safeParse(req.body);
  if (!parsed.success) {
    res.status(400).json({ error: "invalid body" });
    return;
  }
  const user = store.findUserById(userId(req))!;
  const meta: SharedFileMeta = {
    id: randomUUID(),
    owner: user.username,
    name: parsed.data.name,
    mime: parsed.data.mime,
    size: Math.floor((parsed.data.contentBase64.length * 3) / 4),
    updatedAt: new Date().toISOString(),
  };
  files.set(meta.id, { meta, content: parsed.data.contentBase64 });
  res.status(201).json(meta);
});

app.delete("/api/files/:id", authOnly, (req, res) => {
  const stored = files.get(req.params.id);
  if (!stored) {
    res.status(404).json({ error: "not found" });
    return;
  }
  const user = store.findUserById(userId(req))!;
  if (stored.meta.owner !== user.username) {
    res.status(403).json({ error: "forbidden" });
    return;
  }
  files.delete(req.params.id);
  res.status(204).end();
});
```
