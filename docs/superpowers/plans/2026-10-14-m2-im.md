```ts
  z.object({
    type: z.literal("direct"),
    id: z.string().min(1),
    peer: UserSchema,
    unreadCount: z.number().int().min(0),
  }),
  z.object({
    type: z.literal("group"),
    id: z.string().min(1),
    name: z.string().min(1).max(64),
    memberCount: z.number().int().min(2),
    unreadCount: z.number().int().min(0),
  }),
]);
export type Conversation = z.infer<typeof ConversationSchema>;

export const DirectConversationBodySchema = z.object({ peerId: z.string().min(1) });
export const GroupConversationBodySchema = z.object({
  name: z.string().min(1).max(32),
  memberIds: z.array(z.string().min(1)).min(1).max(200),
});
export const CreateConversationBodySchema = z.union([
  DirectConversationBodySchema,
  GroupConversationBodySchema,
]);

export const ReadStateSchema = z.object({ userId: z.string().min(1), at: z.string() });
export type ReadState = z.infer<typeof ReadStateSchema>;

export const MemberSchema = z.object({
  kind: z.enum(["user", "agent"]),
  id: z.string().min(1),
  name: z.string().min(1),
  avatar: z.string().max(600_000).nullable(),
  user: UserSchema.nullable(),
});
export type Member = z.infer<typeof MemberSchema>;
```

```ts
      mustUser(creatorId);
      const ids = [...new Set([creatorId, ...memberIds])];
      for (const id of ids) mustUser(id);
      const id = randomUUID();
      conversations.set(id, { members: ids, kind: "group", name });
      messages.set(id, []);
      reads.set(id, new Map());
      return { id, type: "group", name };
    },
    listConversations(userId) {
      const unread = (convId: string): number => {
        const readAt = reads.get(convId)?.get(userId);
        return (messages.get(convId) ?? []).filter(
          (m) => m.senderId !== userId && (!readAt || m.createdAt > readAt),
        ).length;
      };
      const out: Conversation[] = [];
      for (const [id, conv] of conversations) {
        if (!conv.members.includes(userId)) continue;
        if (conv.kind === "group") {
          out.push({ id, type: "group", name: conv.name!, memberCount: conv.members.length, unreadCount: unread(id) });
        } else {
          const peerId = conv.members.find((m) => m !== userId);
          if (peerId) out.push({ id, type: "direct", peer: mustUser(peerId), unreadCount: unread(id) });
        }
      }
      return out;
    },
    listMembers(conversationId) {
      const conv = conversations.get(conversationId);
      if (!conv) throw new Error("conversation not found");
      return conv.members.map((id) => {
        const u = mustUser(id);
        return { kind: "user", id, name: u.displayName, avatar: u.avatar ?? null, user: u };
      });
    },
    markRead(conversationId, userId) {
      const conv = conversations.get(conversationId);
      if (!conv) throw new Error("conversation not found");
      if (!conv.members.includes(userId)) throw new Error("not a member");
      let m = reads.get(conversationId);
      if (!m) { m = new Map(); reads.set(conversationId, m); }
      m.set(userId, new Date().toISOString());

      if (!m) return [];
```

```ts
import { DatabaseSync } from "node:sqlite";
import { mkdirSync } from "node:fs";
import { dirname } from "node:path";

export function resolveDbPath(): string {
  return process.env.HEMING_DB_PATH?.trim() || "./data/heming.db";
}

export function openDb(dbPath: string): DatabaseSync {
  if (dbPath !== ":memory:") mkdirSync(dirname(dbPath), { recursive: true });
  const db = new DatabaseSync(dbPath);
  db.exec("PRAGMA journal_mode = WAL;");
  db.exec("PRAGMA foreign_keys = ON;");
  initSchema(db);
  return db;
}

export function initSchema(db: DatabaseSync): void {
  db.exec(`
    CREATE TABLE IF NOT EXISTS users (
      id TEXT PRIMARY KEY, username TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL,
      display_name TEXT NOT NULL, avatar TEXT
    );
    CREATE TABLE IF NOT EXISTS friend_requests (
      id TEXT PRIMARY KEY, from_id TEXT NOT NULL, to_id TEXT NOT NULL, status TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS conversations (
      id TEXT PRIMARY KEY, kind TEXT NOT NULL, name TEXT, created_by TEXT
    );
    CREATE TABLE IF NOT EXISTS conversation_members (
      conv_id TEXT NOT NULL, member_id TEXT NOT NULL, PRIMARY KEY (conv_id, member_id)
    );
    CREATE TABLE IF NOT EXISTS messages (
      id TEXT PRIMARY KEY, conv_id TEXT NOT NULL, sender_type TEXT NOT NULL, sender_id TEXT NOT NULL,
      payload TEXT NOT NULL, reply_to TEXT, created_at TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS message_reads (
      conv_id TEXT NOT NULL, user_id TEXT NOT NULL, read_at TEXT NOT NULL,
      PRIMARY KEY (conv_id, user_id)
    );
    CREATE TABLE IF NOT EXISTS agents (
      id TEXT PRIMARY KEY, owner_id TEXT, name TEXT NOT NULL, avatar TEXT,
      description TEXT NOT NULL DEFAULT '', system_prompt TEXT NOT NULL DEFAULT ''
    );
    CREATE INDEX IF NOT EXISTS idx_messages_conv ON messages(conv_id, created_at);
  `);
}
```

- [ ] **Step 5: 实现 `sqliteStore.ts`**

```ts
import { randomUUID } from "node:crypto";
import type { DatabaseSync } from "node:sqlite";
import type { User, Message, Conversation, Member, ReadState } from "@heming/shared";
import { openDb } from "./db.js";

export type SqliteStore = Store & { close(): void };

function rowToUser(r: { id: string; username: string; display_name: string; avatar: string | null }): User {
  return { id: r.id, username: r.username, displayName: r.display_name, ...(r.avatar ? { avatar: r.avatar } : {}) };
}

export function createSqliteStore(dbPath: string): SqliteStore {
  const db: DatabaseSync = openDb(dbPath);

  const mustUser = (id: string): User => {
    const row = db.prepare("SELECT id, username, display_name, avatar FROM users WHERE id = ?").get(id) as
      | { id: string; username: string; display_name: string; avatar: string | null }
      | undefined;
    if (!row) throw new Error("user not found");
    return rowToUser(row);
  };

  return {
    close() { db.close(); },

    createUser(username, passwordHash, displayName) {
      const id = randomUUID();
      try {
        db.prepare("INSERT INTO users (id, username, password_hash, display_name) VALUES (?, ?, ?, ?)")
          .run(id, username, passwordHash, displayName);
      } catch (e) {
        if (String(e).includes("UNIQUE")) throw new Error("username taken");
        throw e;
      }
      return { id, username, displayName };
    },
    findUserByUsername(username) {
      const row = db.prepare(
        "SELECT id, username, display_name, avatar, password_hash FROM users WHERE username = ?",
      ).get(username) as | { id: string; username: string; display_name: string; avatar: string | null; password_hash: string } | undefined;
      return row ? { user: rowToUser(row), passwordHash: row.password_hash } : null;
    },
    findUserById(id) {
      const row = db.prepare("SELECT id, username, display_name, avatar FROM users WHERE id = ?").get(id) as
        | { id: string; username: string; display_name: string; avatar: string | null } | undefined;
      return row ? rowToUser(row) : null;
    },
    updateProfile(userId, patch) {
      const existing = db.prepare("SELECT id, username, display_name, avatar FROM users WHERE id = ?").get(userId) as
        | { id: string; username: string; display_name: string; avatar: string | null } | undefined;
      if (!existing) throw new Error("user not found");
      const nextAvatar = patch.avatar !== undefined ? patch.avatar : existing.avatar;
      db.prepare("UPDATE users SET display_name = ?, avatar = ? WHERE id = ?")
        .run(patch.displayName, nextAvatar, userId);
      return { id: existing.id, username: existing.username, displayName: patch.displayName, ...(nextAvatar ? { avatar: nextAvatar } : {}) };
    },
    createFriendRequest(fromId, toUsername) {
      mustUser(fromId);
      const to = this.findUserByUsername(toUsername);
      if (!to) throw new Error("peer not found");
      if (to.user.id === fromId) throw new Error("cannot friend self");
      const dup = db.prepare(
        `SELECT id FROM friend_requests WHERE status = 'pending' AND
          ((from_id = ? AND to_id = ?) OR (from_id = ? AND to_id = ?))`,
      ).get(fromId, to.user.id, to.user.id, fromId);
      if (dup) throw new Error("request already exists");
      const id = randomUUID();
      db.prepare("INSERT INTO friend_requests (id, from_id, to_id, status) VALUES (?, ?, ?, 'pending')")
        .run(id, fromId, to.user.id);
      return { id, from: mustUser(fromId), to: to.user, status: "pending" };
    },
    acceptFriendRequest(requestId, userId) {
      const row = db.prepare("SELECT id, from_id, to_id, status FROM friend_requests WHERE id = ?").get(requestId) as
        | { id: string; from_id: string; to_id: string; status: string } | undefined;
      if (!row) throw new Error("request not found");
      if (row.to_id !== userId) throw new Error("not request recipient");
      db.prepare("UPDATE friend_requests SET status = 'accepted' WHERE id = ?").run(requestId);
      return { id: row.id, user: mustUser(row.from_id) };
    },
    listFriends(userId) {
      const rows = db.prepare(
        `SELECT id, from_id, to_id FROM friend_requests WHERE status = 'accepted' AND (from_id = ? OR to_id = ?)`,
      ).all(userId, userId) as { id: string; from_id: string; to_id: string }[];
      return rows.map((r) => ({ id: r.id, user: mustUser(r.from_id === userId ? r.to_id : r.from_id) }));
    },
    listIncomingRequests(userId) {
      const rows = db.prepare(
        `SELECT id, from_id FROM friend_requests WHERE status = 'pending' AND to_id = ?`,
      ).all(userId) as { id: string; from_id: string }[];
      return rows.map((r) => ({ id: r.id, from: mustUser(r.from_id) }));
    },
    createDirectConversation(aId, bId) {
      mustUser(aId); mustUser(bId);
      const friends = db.prepare(
        `SELECT 1 FROM friend_requests WHERE status = 'accepted' AND
          ((from_id = ? AND to_id = ?) OR (from_id = ? AND to_id = ?))`,
      ).get(aId, bId, bId, aId);
      if (!friends) throw new Error("not friends");
      const key = [aId, bId].sort().join("|");
      const found = db.prepare(
        `SELECT c.id FROM conversations c
         JOIN conversation_members m1 ON m1.conv_id = c.id AND m1.member_id = ?
         JOIN conversation_members m2 ON m2.conv_id = c.id AND m2.member_id = ?
         WHERE c.kind = 'direct'`,
      ).get(aId, bId) as { id: string } | undefined;
      if (found) return { id: found.id, type: "direct", peer: mustUser(bId) };
      void key;
      const id = randomUUID();
      db.prepare("INSERT INTO conversations (id, kind) VALUES (?, 'direct')").run(id);
      const ins = db.prepare("INSERT INTO conversation_members (conv_id, member_id) VALUES (?, ?)");
      ins.run(id, aId); ins.run(id, bId);
      return { id, type: "direct", peer: mustUser(bId) };
    },
    createGroup(creatorId, name, memberIds) {
      mustUser(creatorId);
      const ids = [...new Set([creatorId, ...memberIds])];
      for (const id of ids) mustUser(id);
      const id = randomUUID();
      db.prepare("INSERT INTO conversations (id, kind, name, created_by) VALUES (?, 'group', ?, ?)").run(id, name, creatorId);
      const ins = db.prepare("INSERT INTO conversation_members (conv_id, member_id) VALUES (?, ?)");
      for (const m of ids) ins.run(id, m);
      return { id, type: "group", name };
    },
    isMember(conversationId, userId) {
      const row = db.prepare("SELECT 1 AS x FROM conversation_members WHERE conv_id = ? AND member_id = ?")
        .get(conversationId, userId);
      return !!row;
    },
    addMessage(conversationId, senderType, senderId, payload, replyTo) {
      if (!this.isMember(conversationId, senderId) && !(await_exists_marker)) { /* 占位说明：直接存在性检查 */ }
      const conv = db.prepare("SELECT id FROM conversations WHERE id = ?").get(conversationId);
      if (!conv) throw new Error("conversation not found");
      const msg: Message = {
        id: randomUUID(), conversationId, senderType, senderId, payload, replyTo,
        createdAt: new Date().toISOString(),
      };
      db.prepare(
        "INSERT INTO messages (id, conv_id, sender_type, sender_id, payload, reply_to, created_at) VALUES (?, ?, ?, ?, ?, ?, ?)",
      ).run(msg.id, conversationId, senderType, senderId, JSON.stringify(payload), replyTo, msg.createdAt);
      return msg;
    },
    listMessages(conversationId, limit) {
      const rows = (limit
        ? db.prepare("SELECT * FROM messages WHERE conv_id = ? ORDER BY created_at DESC, rowid DESC LIMIT ?").all(conversationId, limit)
        : db.prepare("SELECT * FROM messages WHERE conv_id = ? ORDER BY created_at ASC, rowid ASC").all(conversationId)
      ) as { id: string; conv_id: string; sender_type: string; sender_id: string; payload: string; reply_to: string | null; created_at: string }[];
      const ordered = limit ? [...rows].reverse() : rows;
      return ordered.map((r) => ({
        id: r.id, conversationId: r.conv_id,
        senderType: r.sender_type as "user" | "agent", senderId: r.sender_id,
        payload: JSON.parse(r.payload), replyTo: r.reply_to, createdAt: r.created_at,
      }));
    },
    listConversations(userId) {
      const convs = db.prepare(
        `SELECT c.id, c.kind, c.name FROM conversations c
         JOIN conversation_members m ON m.conv_id = c.id WHERE m.member_id = ?`,
      ).all(userId) as { id: string; kind: string; name: string | null }[];
      const unreadOf = (convId: string): number => {
        const read = db.prepare("SELECT read_at FROM message_reads WHERE conv_id = ? AND user_id = ?")
          .get(convId, userId) as { read_at: string } | undefined;
        if (read) {
          const row = db.prepare(
            "SELECT COUNT(*) AS n FROM messages WHERE conv_id = ? AND sender_id != ? AND created_at > ?",
          ).get(convId, userId, read.read_at) as { n: number };
          return Number(row.n);
        }
        const row = db.prepare(
          "SELECT COUNT(*) AS n FROM messages WHERE conv_id = ? AND sender_id != ?",
        ).get(convId, userId) as { n: number };
        return Number(row.n);
      };
      const out: Conversation[] = [];
      for (const c of convs) {
        if (c.kind === "group") {
          const mc = db.prepare("SELECT COUNT(*) AS n FROM conversation_members WHERE conv_id = ?").get(c.id) as { n: number };
          out.push({ id: c.id, type: "group", name: c.name ?? "未命名群", memberCount: Number(mc.n), unreadCount: unreadOf(c.id) });
        } else {
          const peer = db.prepare(
            "SELECT member_id FROM conversation_members WHERE conv_id = ? AND member_id != ?",
          ).get(c.id, userId) as { member_id: string } | undefined;
          if (peer) out.push({ id: c.id, type: "direct", peer: mustUser(peer.member_id), unreadCount: unreadOf(c.id) });
        }
      }
      return out;
    },
    listMembers(conversationId) {
      const rows = db.prepare("SELECT member_id FROM conversation_members WHERE conv_id = ?").all(conversationId)
        as { member_id: string }[];
      return rows.map((r) => {
        const u = mustUser(r.member_id);
        return { kind: "user" as const, id: r.member_id, name: u.displayName, avatar: u.avatar ?? null, user: u };
      });
    },
    markRead(conversationId, userId) {
      if (!this.isMember(conversationId, userId)) throw new Error("not a member");
      db.prepare(
        `INSERT INTO message_reads (conv_id, user_id, read_at) VALUES (?, ?, ?)
         ON CONFLICT(conv_id, user_id) DO UPDATE SET read_at = excluded.read_at`,
      ).run(conversationId, userId, new Date().toISOString());
    },
    listReads(conversationId) {
      const rows = db.prepare("SELECT user_id, read_at FROM message_reads WHERE conv_id = ?").all(conversationId)
        as { user_id: string; read_at: string }[];
      return rows.map((r) => ({ userId: r.user_id, at: r.read_at }));
    },
  };
}
```

```tsx
import { useState } from "react";
import { useChatStore } from "../store/chat.js";

export function GroupCreateModal({ open, onClose }: { open: boolean; onClose: () => void }) {
  const friends = useChatStore((s) => s.friends);
  const createGroup = useChatStore((s) => s.createGroup);
  const [name, setName] = useState("");
  const [picked, setPicked] = useState<string[]>([]);
  const [error, setError] = useState("");
  const [pending, setPending] = useState(false);
  if (!open) return null;

  const toggle = (id: string) =>
    setPicked((p) => (p.includes(id) ? p.filter((x) => x !== id) : [...p, id]));

  const submit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!name.trim() || pending) { setError("请输入群名称"); return; }
    setPending(true);
    setError("");
    try {
      await createGroup(name.trim(), picked);
      setName(""); setPicked([]);
      onClose();
    } catch (err) {
      setError((err as Error).message);
    } finally {
      setPending(false);
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/20">
      <form onSubmit={submit} className="w-[320px] rounded-xl border border-line bg-white p-5 shadow-card">
        <h3 className="mb-3 text-[14px] font-bold">新建群聊</h3>
        <input
          className="mb-3 w-full rounded-lg border border-line px-3 py-2 text-sm outline-none focus:border-[#bbb]"
          placeholder="群名称" maxLength={32} value={name}
          onChange={(e) => setName(e.target.value)}
        />
        <div className="mb-3 max-h-[180px] overflow-y-auto">
          {friends.length === 0 ? (
            <div className="text-[12px] text-muted">暂无好友可邀请</div>
          ) : (
            friends.map((f) => (
              <label key={f.id} className="flex items-center gap-2 py-1 text-[13px]">
                <input
                  type="checkbox"
                  aria-label={`选择成员 ${f.user.displayName}`}
                  checked={picked.includes(f.user.id)}
                  onChange={() => toggle(f.user.id)}
                />
                {f.user.displayName}
              </label>
            ))
          )}
        </div>
        {error && <div className="mb-2 text-[12px] text-red-600">{error}</div>}
        <div className="flex justify-end gap-2">
          <button type="button" onClick={onClose} className="rounded-lg border border-line px-3 py-1.5 text-[13px]">取消</button>
          <button type="submit" className="rounded-lg bg-ink px-3 py-1.5 text-[13px] font-bold text-white">创建</button>
        </div>
      </form>
    </div>
  );
}
```
